# v2.1 Lightbox Fix + Mobile Zoom Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fix lightbox map-state reset and nearby-group recomputation on every prev/next press, and fix Leaflet zoom controls being too small for mobile touch.

**Architecture:** Merge `MainMapView` and `ImageView` into a single `MapPage` layout-route component that stays mounted across `/` and `/image/:imageId`, preserving Leaflet map state naturally. Nearby group is computed once per modal open session and locked in state. CSS overrides fix Leaflet zoom control touch targets.

**Tech Stack:** React 19, React Router v7 (layout routes), react-leaflet v5, Leaflet, TypeScript (strict), Jest + ts-jest (tests), ESLint

- [ ] **Step 3: Create `packages/frontend/src/utils/nearbyGroup.ts`**

  ```ts
  import type { ImageRecord } from '../types/index.js';
  import { calculateHaversineDistance } from './distance.js';

  /**
   * Compute the navigation group for a modal open session.
   * Returns an array with `selected` first, followed by all images within
   * `radiusKm` kilometres sorted by distance ascending.
   * The group is locked for the duration of the modal session — it must NOT
   * be recomputed on prev/next navigation.
   */
  export function computeNearbyGroup(
    selected: ImageRecord,
    allImages: ImageRecord[],
    radiusKm = 1,
  ): ImageRecord[] {
    const nearby = allImages
      .filter(img => img.id !== selected.id)
      .map(img => ({
        image: img,
        distance: calculateHaversineDistance(
          selected.latitude,
          selected.longitude,
          img.latitude,
          img.longitude,
        ),
      }))
      .filter(item => item.distance <= radiusKm)
      .sort((a, b) => a.distance - b.distance)
      .map(item => item.image);

    return [selected, ...nearby];
  }
  ```

## Chunk 3: MapPage component

### Task 4: Create `MapPage.tsx` and rename `MainMapView.css`

**Files:**
- Create: `packages/frontend/src/components/MapPage.tsx`
- Rename: `packages/frontend/src/components/MainMapView.css` → `packages/frontend/src/components/MapPage.css`
  (class names inside the file are **not** changed — `.main-map-view` is retained)

`MapPage` replaces both `MainMapView` and `ImageView`. It is the layout-route parent mounted for all routes. It holds:
- `images`, `loading`, `error` (same as current `MainMapView`)
- `selectedImage`, `nearbyGroup` (locked per modal session)
- `mapRef` (useRef — does **not** trigger re-renders)
- `pendingPanRef` (useRef — stores coordinates to pan once map is ready; avoids stale closure in `onMapReady`)

> **Note:** `pendingPan` is implemented as `useRef` (not `useState`) so that the `onMapReady` callback can use `useCallback([])` (stable reference) while still reading the latest pending pan value. A `useState` field would be stale inside a `useCallback([])` closure.

- [ ] **Step 2: Create `MapPage.tsx`**

  ```tsx
  import L from 'leaflet';
  import type React from 'react';
  import { useCallback, useEffect, useRef, useState } from 'react';
  import { useNavigate, useParams } from 'react-router-dom';
  import { apiClient } from '../services';
  import type { ImageRecord } from '../types';
  import { extractImageId } from '../utils/urlUtils';
  import { computeNearbyGroup } from '../utils/nearbyGroup';
  import ImageModal from './ImageModal';
  import MapView from './MapView';
  import './MapPage.css';

  const MapPage: React.FC = () => {
    const navigate = useNavigate();
    const { imageId: rawImageId } = useParams<{ imageId: string }>();

    const [images, setImages] = useState<ImageRecord[]>([]);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState<string | null>(null);
    const [selectedImage, setSelectedImage] = useState<ImageRecord | null>(null);
    const [nearbyGroup, setNearbyGroup] = useState<ImageRecord[]>([]);

    // useRef fields — mutations do NOT trigger re-renders
    const mapRef = useRef<L.Map | null>(null);
    const pendingPanRef = useRef<[number, number] | null>(null);

    // Load images once on mount; handle deep-link if imageId is present in the URL
    useEffect(() => {
      const loadImages = async () => {
        try {
          setLoading(true);
          setError(null);
          const loadedImages = await apiClient.getAllImages();
          setImages(loadedImages);

          if (rawImageId) {
            const cleanId = extractImageId(rawImageId);
            const image = loadedImages.find(img => img.id === cleanId);

            if (!image) {
              navigate('/', { replace: true });
              return;
            }

            const group = computeNearbyGroup(image, loadedImages);
            setSelectedImage(image);
            setNearbyGroup(group);

            // Pan to image — defer via pendingPanRef if map isn't mounted yet
            if (mapRef.current) {
              mapRef.current.panTo([image.latitude, image.longitude]);
            } else {
              pendingPanRef.current = [image.latitude, image.longitude];
            }
          }
        } catch (err) {
          console.error('Failed to load images:', err);
          setError(err instanceof Error ? err.message : 'Failed to load images');
        } finally {
          setLoading(false);
        }
      };

      loadImages();
    // eslint-disable-next-line react-hooks/exhaustive-deps
    }, []); // runs once on mount — rawImageId is intentionally not a dep (deep-link only)

    // Stable callback — safe with useCallback([]) because pendingPanRef is a ref (not state)
    const onMapReady = useCallback((map: L.Map) => {
      mapRef.current = map;
      if (pendingPanRef.current) {
        map.panTo(pendingPanRef.current);
        pendingPanRef.current = null;
      }
    }, []);

    const handleImageSelect = useCallback((imageId: string) => {
      const cleanId = extractImageId(imageId);
      const image = images.find(img => img.id === cleanId);
      if (!image) return;
      const group = computeNearbyGroup(image, images);
      setSelectedImage(image);
      setNearbyGroup(group);
      navigate(`/image/${image.id}`);
    }, [images, navigate]);

    const handleNavigate = useCallback((direction: 'prev' | 'next') => {
      if (!selectedImage || nearbyGroup.length <= 1) return;
      const currentIndex = nearbyGroup.findIndex(img => img.id === selectedImage.id);
      const nextIndex = direction === 'prev'
        ? (currentIndex > 0 ? currentIndex - 1 : nearbyGroup.length - 1)
        : (currentIndex < nearbyGroup.length - 1 ? currentIndex + 1 : 0);
      const nextImage = nearbyGroup[nextIndex];
      setSelectedImage(nextImage);
      navigate(`/image/${nextImage.id}`, { replace: true });
    }, [selectedImage, nearbyGroup, navigate]);

    const handleCloseModal = useCallback(() => {
      setSelectedImage(null);
      setNearbyGroup([]);
      navigate('/');
    }, [navigate]);

    if (loading) {
      return (
        <div className="main-map-view">
          <div className="loading-container">
            <div className="loading-spinner"></div>
            <p>Loading images...</p>
          </div>
        </div>
      );
    }

    if (error) {
      return (
        <div className="main-map-view">
          <div className="error-container">
            <h2>Error Loading Images</h2>
            <p>{error}</p>
            <button type="button" onClick={() => window.location.reload()}>
              Retry
            </button>
          </div>
        </div>
      );
    }

    return (
      <div className="main-map-view">
        <header className="map-header">
          <h1>Spyo Map</h1>
          <div className="image-count">{images.length} geo-tagged images</div>
        </header>

        <div className="map-wrapper">
          <MapView
            images={images}
            selectedImageId={selectedImage?.id}
            onImageSelect={handleImageSelect}
            onMapReady={onMapReady}
          />
        </div>

        <ImageModal
          image={selectedImage}
          nearbyImages={nearbyGroup}
          isOpen={!!selectedImage}
          onClose={handleCloseModal}
          onNavigate={handleNavigate}
        />
      </div>
    );
  };

  export default MapPage;
  ```
